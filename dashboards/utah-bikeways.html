rows = data
  .filter(d => d.Name && d.Name !== "NA" && d["All Roads"] > 0)
  .map(d => {
    const bike = bikeTypes.reduce((a, t) => a + (d[t] || 0), 0);
    return {...d, bike, ratio: bike / d["All Roads"] * 100};
  })
avgRatio = d3.mean(rows, d => d.ratio)
rowsTagged = rows.map(d => ({...d, group: d.ratio >= avgRatio ? "Above average" : "Below average"}))
highlighted = rowsTagged.filter(d => searchCity && d.Name.toLowerCase().includes(searchCity.toLowerCase()))
totalBike = d3.sum(rows, d => d.bike)
totalRoad = d3.sum(rows, d => d["All Roads"])
top = d3.greatest(rows.filter(d => d["All Roads"] > 100), d => d.ratio)
